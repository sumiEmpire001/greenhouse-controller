cmake_minimum_required(VERSION 3.10)
project(greenhouse_cpp LANGUAGES CXX)
enable_testing()
add_executable(control_tests cpp/tests/control_tests.cpp)
target_include_directories(control_tests PRIVATE arduino/GreenhouseController/src)
set_target_properties(control_tests PROPERTIES CXX_STANDARD 11 CXX_STANDARD_REQUIRED YES)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(control_tests PRIVATE -Wall -Wextra -Werror -pedantic)
endif()
add_test(NAME greenhouse_control COMMAND control_tests)
